  6/-!
  7# The Hoeffding bound on the running-maximum gap (multigen note, Proposition 5, first bound)
  8
  9Source: `paper/evaluation_feedback_multigen.md`, draft 0.5, §2, Proposition 5; code
 10`src/spiral_ln/evaluation_feedback_multigen.py` (`naive_gap_bounds`).
 11
 12**Proposition 5 (first bound).** For every `n ≥ 1`, `K`, `p ∈ [0, 1]` and `G`, with `m = G K + 1`,
 13the naive readout gap satisfies `ρ_G ≤ √(ln m/(2n))`.
 14
 15*Proof (as in the note).* `Y_i = X_i - np` is a sum of `n` centred Bernoullis; Hoeffding's lemma
 16(Mathlib's `ProbabilityTheory.hasSubgaussianMGF_of_mem_Icc`) and independence give
 17`E e^{λ Y_i} ≤ e^{λ² n/8}` (`binomial_mgf_le`); the maximal inequality
 18`E max_i Y_i ≤ λ⁻¹ ln ∑_i E e^{λ Y_i}` (Jensen, then `max ≤ ∑`; `maximal_inequality`) gives
 19`(ln m)/λ + λ n/8`, which `λ = √(8 ln m/n)` makes `√(n ln m/2)`; divide by `n`.
 20
 21The draws are the `G K + 1` i.i.d. `Bin(n, p)` counts of `MultigenRunningMax` (product law over
 22`Draw G K`). The second (Bernstein) bound of Proposition 5 is not formalized.
 23-/

 30/-! ## Hoeffding's lemma for one Bernoulli, from Mathlib -/
 31
 32/-- **Hoeffding's lemma** for `B ~ Bernoulli(p)`: `E e^{t(B - p)} ≤ e^{t²/8}`, obtained from
 33Mathlib's sub-Gaussian form of Hoeffding's lemma on the two-point law. -/
 34theorem bernoulli_hoeffding {p : ℝ} (hp0 : 0 ≤ p) (hp1 : p ≤ 1) (t : ℝ) :
 35    p * exp (t * (1 - p)) + (1 - p) * exp (t * (0 - p)) ≤ exp (t ^ 2 / 8) := by
 36  set q : ℝ≥0 := p.toNNReal with hqdef
 37  have hq : q ≤ 1 := Real.toNNReal_le_one.mpr hp1
 38  have hqp : (q : ℝ) = p := Real.coe_toNNReal p hp0
 39  set μ : Measure Bool := (PMF.bernoulli q hq).toMeasure with hμ
 40  let X : Bool → ℝ := fun b => if b then 1 else 0
 41  have hX : AEMeasurable X μ := (measurable_of_countable X).aemeasurable
 42  have hb : ∀ᵐ b ∂μ, X b ∈ Set.Icc (0 : ℝ) 1 :=
 43    Filter.Eventually.of_forall fun b => by cases b <;> simp [X]
 44  have hsg := hasSubgaussianMGF_of_mem_Icc hX hb
 45  have hle := hsg.mgf_le t
 46  have hint : ∀ f : Bool → ℝ, ∫ b, f b ∂μ = p * f true + (1 - p) * f false := by
 47    intro f
 48    rw [hμ, PMF.integral_eq_sum]
 49    simp only [Fintype.univ_bool, Finset.mem_singleton, Bool.true_eq_false, not_false_eq_true,
 50      Finset.sum_insert, Finset.sum_singleton, PMF.bernoulli_apply, Bool.cond_true, Bool.cond_false,
 51      smul_eq_mul]
 52    rw [ENNReal.coe_toReal, ENNReal.coe_toReal, NNReal.coe_sub hq, hqp, NNReal.coe_one]
 53  have hmean : ∫ b, X b ∂μ = p := by rw [hint]; simp [X]
 54  unfold mgf at hle
 55  rw [hint, hmean] at hle
 56  simp only [X, ite_true, Bool.false_eq_true, ite_false] at hle
 57  have hc : (((‖(1 : ℝ) - 0‖₊ / 2) ^ 2 : ℝ≥0) : ℝ) = 1 / 4 := by
 58    simp
 59    norm_num
 60  rw [hc] at hle
 61  calc p * exp (t * (1 - p)) + (1 - p) * exp (t * (0 - p)) ≤ exp (1 / 4 * t ^ 2 / 2) := hle
 62    _ = exp (t ^ 2 / 8) := by ring_nf

 64/-! ## The binomial mgf -/
 65
 66/-- `E e^{t(X - np)} = (p e^{t(1-p)} + (1-p) e^{-tp})^n` for `X ~ Bin(n, p)`. -/
 67theorem binomial_mgf (n : ℕ) (p t : ℝ) :
 68    ∑ x : Fin (n + 1), binomial n p x * exp (t * ((x : ℝ) - n * p)) =
 69      (p * exp (t * (1 - p)) + (1 - p) * exp (t * (0 - p))) ^ n := by
 70  rw [add_pow, ← Fin.sum_univ_eq_sum_range]
 71  apply Finset.sum_congr rfl
 72  intro x _
 73  have hx : x.val ≤ n := Nat.lt_succ_iff.mp x.isLt
 74  unfold binomial
 75  rw [mul_pow, mul_pow, ← Real.exp_nat_mul, ← Real.exp_nat_mul, Nat.cast_sub hx]
 76  have : rexp (t * ((x : ℝ) - n * p)) =
 77      rexp (x * (t * (1 - p))) * rexp (((n : ℝ) - x) * (t * (0 - p))) := by
 78    rw [← Real.exp_add]; congr 1; ring
 79  rw [this]
 80  ring

 82/-- **Hoeffding for one count:** `E e^{t(X - np)} ≤ e^{n t²/8}` for `X ~ Bin(n, p)`. -/
 83theorem binomial_mgf_le (n : ℕ) {p : ℝ} (hp0 : 0 ≤ p) (hp1 : p ≤ 1) (t : ℝ) :
 84    ∑ x : Fin (n + 1), binomial n p x * exp (t * ((x : ℝ) - n * p)) ≤ exp (n * t ^ 2 / 8) := by
 85  rw [binomial_mgf]
 86  have h := bernoulli_hoeffding hp0 hp1 t
 87  have h0 : 0 ≤ p * exp (t * (1 - p)) + (1 - p) * exp (t * (0 - p)) := by
 88    have : 0 ≤ 1 - p := by linarith
 89    positivity
 90  calc (p * exp (t * (1 - p)) + (1 - p) * exp (t * (0 - p))) ^ n ≤ exp (t ^ 2 / 8) ^ n :=
 91        pow_le_pow_left₀ h0 h n
 92    _ = exp (n * t ^ 2 / 8) := by rw [← Real.exp_nat_mul]; ring_nf

100/-- One coordinate of i.i.d. draws has law `w`. -/
101theorem expect_coord {ι : Type*} [Fintype ι] [DecidableEq ι] (hw : ∑ x, w x = 1) (i : ι)
102    (f : Fin (n + 1) → ℝ) :
103    MultigenRunningMax.expect w (fun D : ι → Fin (n + 1) => f (D i)) = ∑ x, w x * f x := by
104  unfold MultigenRunningMax.expect
105  have hprod : ∀ D : ι → Fin (n + 1), (∏ j, w (D j)) * f (D i) =
106      ∏ j, (if j = i then w (D j) * f (D j) else w (D j)) := by
107    intro D
108    have : ∀ j, (if j = i then w (D j) * f (D j) else w (D j)) =
109        w (D j) * (if j = i then f (D j) else 1) := by
110      intro j; split_ifs <;> ring
111    simp_rw [this, Finset.prod_mul_distrib, Finset.prod_ite_eq', ite_eq_left (mem_univ i)]
112  simp_rw [hprod]
113  have h := Finset.prod_univ_sum (fun _ : ι => (univ : Finset (Fin (n + 1))))
114    (fun j x => if j = i then w x * f x else w x)
115  rw [Fintype.piFinset_univ] at h
116  rw [← h, Finset.prod_eq_single i]
117  · simp
118  · intro j _ hj
119    simp [hj, hw]
120  · intro h'; exact absurd (mem_univ i) h'

122/-- **The maximal inequality** (Jensen, then `max ≤ ∑`): for every real `λ` and `m ≥ 1` i.i.d. draws,
123`λ (E[max_i X_i] - c) ≤ ln m + ln E e^{λ (X - c)}`. -/
124theorem maximal_inequality {ι : Type*} [Fintype ι] [DecidableEq ι] [Nonempty ι]
125    (hw0 : ∀ x, 0 ≤ w x) (hw : ∑ x, w x = 1) (c lam : ℝ) :
126    lam * (MultigenRunningMax.expect w (fun D : ι → Fin (n + 1) => (maxDraw D : ℝ)) - c) ≤
127      log (Fintype.card ι) + log (∑ x, w x * exp (lam * ((x : ℝ) - c))) := by
128  set E := MultigenRunningMax.expect w (fun D : ι → Fin (n + 1) => (maxDraw D : ℝ))
129  have hmass := total_mass w hw (ι := ι)
130  have hweights : ∀ D : ι → Fin (n + 1), 0 ≤ ∏ i, w (D i) :=
131    fun D => Finset.prod_nonneg fun i _ => hw0 _
132  -- Jensen for `exp`
133  have hJ : exp (lam * (E - c)) ≤
134      ∑ D : ι → Fin (n + 1), (∏ i, w (D i)) * exp (lam * ((maxDraw D : ℝ) - c)) := by
135    have hE : lam * (E - c) = ∑ D : ι → Fin (n + 1), (∏ i, w (D i)) *
136        (lam * ((maxDraw D : ℝ) - c)) := by
137      simp only [E, MultigenRunningMax.expect]
138      rw [mul_sub, Finset.mul_sum]
139      have : lam * c = ∑ D : ι → Fin (n + 1), (∏ i, w (D i)) * (lam * c) := by
140        rw [← Finset.sum_mul, hmass, one_mul]
141      rw [this, ← Finset.sum_sub_distrib]
142      apply Finset.sum_congr rfl; intro D _; ring
143    rw [hE]
144    have := convexOn_exp.map_sum_le (t := univ) (w := fun D : ι → Fin (n + 1) => ∏ i, w (D i))
145      (p := fun D => lam * ((maxDraw D : ℝ) - c)) (fun D _ => hweights D) hmass
146      (fun D _ => Set.mem_univ _)
147    simpa [smul_eq_mul] using this
148  -- `exp(λ max) ≤ ∑_i exp(λ Y_i)`
149  have hmax : ∀ D : ι → Fin (n + 1), exp (lam * ((maxDraw D : ℝ) - c)) ≤
150      ∑ i, exp (lam * (((D i : ℕ) : ℝ) - c)) := by
151    intro D
152    obtain ⟨i, _, hi⟩ := Finset.exists_mem_eq_sup (univ : Finset ι) univ_nonempty
153      (fun i => (D i).val)
154    have : (maxDraw D : ℝ) = ((D i : ℕ) : ℝ) := by rw [maxDraw, hi]
155    rw [this]
156    exact Finset.single_le_sum (f := fun i => exp (lam * (((D i : ℕ) : ℝ) - c)))
157      (fun j _ => (exp_pos _).le) (mem_univ i)
158  have hsum : ∑ D : ι → Fin (n + 1), (∏ i, w (D i)) * exp (lam * ((maxDraw D : ℝ) - c)) ≤
159      Fintype.card ι * ∑ x, w x * exp (lam * ((x : ℝ) - c)) := by
160    calc ∑ D : ι → Fin (n + 1), (∏ i, w (D i)) * exp (lam * ((maxDraw D : ℝ) - c))
161        ≤ ∑ D : ι → Fin (n + 1), (∏ i, w (D i)) * ∑ i, exp (lam * (((D i : ℕ) : ℝ) - c)) :=
162          Finset.sum_le_sum fun D _ => mul_le_mul_of_nonneg_left (hmax D) (hweights D)
163      _ = ∑ i : ι, MultigenRunningMax.expect w (fun D : ι → Fin (n + 1) => exp (lam * (((D i : ℕ) : ℝ) - c))) := by
164          unfold MultigenRunningMax.expect
165          simp_rw [Finset.mul_sum]
166          rw [Finset.sum_comm]
167      _ = ∑ _i : ι, ∑ x, w x * exp (lam * ((x : ℝ) - c)) := by
168          apply Finset.sum_congr rfl
169          intro i _
170          exact expect_coord w hw i (fun x => exp (lam * ((x : ℝ) - c)))
171      _ = Fintype.card ι * ∑ x, w x * exp (lam * ((x : ℝ) - c)) := by
172          rw [Finset.sum_const, Finset.card_univ, nsmul_eq_mul]
173  have hpos : 0 < ∑ x, w x * exp (lam * ((x : ℝ) - c)) := by
174    obtain ⟨x, hx⟩ : ∃ x, 0 < w x := by
175      by_contra h
176      have : ∑ x, w x ≤ 0 := Finset.sum_nonpos fun x _ => not_lt.mp fun hx => h ⟨x, hx⟩
177      linarith
178    exact lt_of_lt_of_le (mul_pos hx (exp_pos _))
179      (Finset.single_le_sum (f := fun x => w x * exp (lam * ((x : ℝ) - c)))
180        (fun y _ => mul_nonneg (hw0 y) (exp_pos _).le) (mem_univ x))
181  have hcard : (0 : ℝ) < Fintype.card ι := by exact_mod_cast Fintype.card_pos (α := ι)
182  have := hJ.trans hsum
183  rw [← Real.log_mul hcard.ne' hpos.ne']
184  rw [← Real.log_exp (lam * (E - c))]
185  exact Real.log_le_log (exp_pos _) this

187/-- **Proposition 5 (Hoeffding), for the maximum of `m` i.i.d. `Bin(n, p)` draws:**
188`E[max_i X_i]/n - p ≤ √(ln m/(2n))`. -/
189theorem max_gap_le_hoeffding (hn : 1 ≤ n) {p : ℝ} (hp0 : 0 ≤ p) (hp1 : p ≤ 1)
190    {ι : Type*} [Fintype ι] [DecidableEq ι] [Nonempty ι] :
191    MultigenRunningMax.expect (binomial n p) (fun D : ι → Fin (n + 1) => (maxDraw D : ℝ)) / n - p ≤
192      √(log (Fintype.card ι) / (2 * n)) := by
193  set E := MultigenRunningMax.expect (binomial n p) (fun D : ι → Fin (n + 1) => (maxDraw D : ℝ))
194  set L := log (Fintype.card ι)
195  have hn' : (0 : ℝ) < n := by exact_mod_cast hn
196  have hL : 0 ≤ L := Real.log_nonneg (by exact_mod_cast Fintype.card_pos (α := ι))
197  -- the bound for every `λ > 0`
198  have hbound : ∀ lam : ℝ, 0 < lam → lam * (E - n * p) ≤ L + n * lam ^ 2 / 8 := by
199    intro lam hlam
200    have h1 := maximal_inequality (ι := ι) (binomial n p) (binomial_nonneg n hp0 hp1)
201      (binomial_sum n p) (n * p) lam
202    have h2 : log (∑ x : Fin (n + 1), binomial n p x * exp (lam * ((x : ℝ) - n * p))) ≤
203        n * lam ^ 2 / 8 := by
204      have hpos : 0 < ∑ x : Fin (n + 1), binomial n p x * exp (lam * ((x : ℝ) - n * p)) := by
205        rw [binomial_mgf]
206        have : 0 < p * exp (lam * (1 - p)) + (1 - p) * exp (lam * (0 - p)) := by
207          rcases eq_or_lt_of_le hp0 with h | h
208          · rw [← h]; simp
209          · have : 0 ≤ 1 - p := by linarith
210            have := exp_pos (lam * (1 - p))
211            have := exp_pos (lam * (0 - p))
212            positivity
213        positivity
214      calc log _ ≤ log (exp (n * lam ^ 2 / 8)) :=
215            Real.log_le_log hpos (binomial_mgf_le n hp0 hp1 lam)
216        _ = n * lam ^ 2 / 8 := Real.log_exp _
217    linarith
218  -- optimize over `λ`
219  have hgap : E - n * p ≤ √(n * L / 2) := by
220    rcases eq_or_lt_of_le hL with hL0 | hLpos
221    · rw [← hL0, mul_zero, zero_div, Real.sqrt_zero]
222      by_contra hcon
223      have hpos : 0 < E - n * p := lt_of_not_ge hcon
224      have := hbound (4 * (E - n * p) / n) (by positivity)
225      rw [← hL0] at this
226      field_simp at this
227      nlinarith
228    · set r := √(n * L / 8) with hr
229      have hrpos : 0 < r := Real.sqrt_pos.mpr (by positivity)
230      have hr2 : r ^ 2 = n * L / 8 := Real.sq_sqrt (by positivity)
231      have := hbound (8 * r / n) (by positivity)
232      have hsq : √(n * L / 2) = 2 * r := by
233        rw [hr, show n * L / 2 = 2 ^ 2 * (n * L / 8) by ring, Real.sqrt_mul' _ (by positivity),
234          Real.sqrt_sq (by norm_num)]
235      rw [hsq]
236      have h8 : 8 * r / n * (E - n * p) ≤ L + n * (8 * r / n) ^ 2 / 8 := this
237      have hL' : L = 8 * r ^ 2 / n := by rw [hr2]; field_simp
238      rw [hL'] at h8
239      have : 8 * r / n * (E - n * p) ≤ 8 * r / n * (2 * r) := by
240        have : L + n * (8 * r / n) ^ 2 / 8 = 8 * r / n * (2 * r) := by
241          rw [hL']; field_simp; ring
242        rw [hL'] at this
243        linarith
244      exact le_of_mul_le_mul_left this (by positivity)
245  -- divide by `n`
246  have hsqrt : √(L / (2 * n)) = √(n * L / 2) / n := by
247    have h : √(n * L / 2) = √(L / (2 * n)) * n := by
248      rw [show (n : ℝ) * L / 2 = (L / (2 * n)) * (n : ℝ) ^ 2 by field_simp,
249        Real.sqrt_mul' _ (by positivity), Real.sqrt_sq hn'.le]
250    rw [h]
251    field_simp
252  rw [hsqrt, div_sub' hn'.ne', div_le_div_iff_of_pos_right hn']
253  linarith
254
255end Max

257/-- **Proposition 5 (Hoeffding bound), for the chain:** under naive reuse with `G` generations of
258`K` candidates on `n ≥ 1` items and i.i.d. `Bin(n, p)` counts, the readout gap
259`ρ_G = E[c_G]/n - p` satisfies `ρ_G ≤ √(ln(G K + 1)/(2n))`. -/
260theorem prop5_hoeffding {n : ℕ} (hn : 1 ≤ n) {p : ℝ} (hp0 : 0 ≤ p) (hp1 : p ≤ 1) (G K : ℕ) :
261    MultigenRunningMax.expect (binomial n p) (fun D : Draw G K → Fin (n + 1) => (reported D : ℝ)) / n - p ≤
262      √(log ((G * K + 1 : ℕ) : ℝ) / (2 * n)) := by
263  have h := max_gap_le_hoeffding hn hp0 hp1 (ι := Draw G K)
264  rw [card_draw] at h
265  have : MultigenRunningMax.expect (binomial n p) (fun D : Draw G K → Fin (n + 1) => (reported D : ℝ)) =
266      MultigenRunningMax.expect (binomial n p) (fun D : Draw G K → Fin (n + 1) => (maxDraw D : ℝ)) := by
267    unfold MultigenRunningMax.expect
268    simp only [reported_eq_maxDraw]
269  rw [this]
270  exact h
